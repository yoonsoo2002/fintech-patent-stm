file_path <- "C:/Users/gram/Documents/핀테크 특허 분석/데이터 임베딩완료/final_fintech_patents_filtered.csv"

# 4. 텍스트 결합 (제목 가중치 2배 적용)

[ 조정된 피인용수 계산 ] 
# ==========================================
# 📊 1단계: 조정된 피인용수(C_adj_i) 계산하기
# ==========================================
# (1) 분모 계산: 특허가 출원된 연도(Publication.Year)별 '평균 피인용수(C_avg_j)'를 구합니다.
out3$meta$C_avg_j <- ave(out3$meta$Cited.by.Patent.Count, 
                         out3$meta$Publication.Year, 
                         FUN = function(x) mean(x, na.rm = TRUE))

# (2) 수식 적용: 개별 특허의 피인용수를 해당 연도 평균으로 나눕니다. 
# (만약 특정 연도에 피인용수가 모두 0이라서 평균이 0이 될 경우, 수학적 에러(분모가 0)를 방지하기 위해 0으로 처리합니다.)
out3$meta$C_adj_i <- ifelse(out3$meta$C_avg_j == 0, 
                            0, 
                            out3$meta$Cited.by.Patent.Count / out3$meta$C_avg_j)

# (3) 눈으로 팩트 체크: 계산이 잘 되었는지 상위 10개 특허만 뽑아서 확인해 봅니다.
print("--- 팩트 체크: 연도별 평균 및 조정된 피인용수 계산 결과 ---")
print(head(out3$meta[, c("Publication.Year", "Cited.by.Patent.Count", "C_avg_j", "C_adj_i")], 10))

[ 10~80 탐색 10단위로 ] 
# ==========================================
# 🚀 2단계: 넓은 범위(10~80) 탐색 및 결과 저장 (조정된 피인용수 적용)
# ==========================================
# (작업 폴더는 이미 'k찾기'로 설정되어 있다고 가정합니다)

set.seed(2026) 
# 💡 핵심: prevalence 공식에 log1p(...) 대신 C_adj_i 가 들어갑니다!
kresult_broad_adj <- searchK(documents = out3$documents, 
                             vocab = out3$vocab, 
                             K = seq(10, 80, by = 10), 
                             prevalence = ~ Origin_Country + s(Publication.Year) + C_adj_i, 
                             data = out3$meta, 
                             max.em.its = 75, 
                             init.type = "Spectral")

# [결과 저장용 데이터 정리]
K_broad <- unlist(kresult_broad_adj$results$K)
z_semcoh_broad <- as.numeric(scale(unlist(kresult_broad_adj$results$semcoh)))
z_exclus_broad <- as.numeric(scale(unlist(kresult_broad_adj$results$exclus)))
score_broad <- (z_semcoh_broad + z_exclus_broad) / 2

df_broad_adj <- data.frame(K = K_broad, 
                           Coherence_Z = round(z_semcoh_broad, 3), 
                           Exclusivity_Z = round(z_exclus_broad, 3), 
                           Final_Score = round(score_broad, 3))

# 1등 점수부터 순서대로 정렬
df_broad_adj_sorted <- df_broad_adj[order(-df_broad_adj$Final_Score), ]

# [CSV 표 저장]
write.csv(df_broad_adj_sorted, "Zscore_Table_1_Broad(10_80)_Adj.csv", row.names = FALSE)
print("✅ [표 저장 완료] Zscore_Table_1_Broad(10_80)_Adj.csv 파일 생성!")

# [꺾은선 그래프(PNG) 저장]
png(filename = "Zscore_Plot_1_Broad(10_80)_Adj.png", width = 1200, height = 800, res = 150)
plot(df_broad_adj$K, df_broad_adj$Final_Score, 
     type="b", pch=19, col="red", lwd=2,
     xlab="Number of Topics (K)", 
     ylab="Combined Z-Score",
     main="Optimal K Broad Search (Adj. Citation): Coherence & Exclusivity")
dev.off()
print("🖼️ [그래프 저장 완료] Zscore_Plot_1_Broad(10_80)_Adj.png 파일 생성!")
