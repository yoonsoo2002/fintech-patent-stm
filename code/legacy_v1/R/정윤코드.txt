                       removenumbers = FALSE,

# ==========================================
hc_topics <- hclust(topic_dist, method = "complete")
# method는 "complete", "average", "ward.D2" 등 바꿔볼 수 있음
# 논문/그림 느낌상 complete 또는 average를 많이 씀

# ==========================================
# 4. 2개 군집 / 4개 군집 cut 높이 계산
# ==========================================
get_cut_height <- function(hc, k){
  n <- length(hc$order)
  if(k < 2 || k > n){
    stop("k는 2 이상, 토픽 수 이하만 가능합니다.")
  }
  # hclust$height는 병합 높이
  # k개 군집으로 자를 때 기준 높이
  sort(hc$height, decreasing = TRUE)[k - 1]
}

# prevalence를 보기 좋은 크기로 스케일링
scale_size <- function(x, min_size = 12, max_size = 35) {
  if (max(x) == min(x)) return(rep(mean(c(min_size, max_size)), length(x)))
  min_size + (x - min(x)) * (max_size - min_size) / (max(x) - min(x))
}

# 5. estimateEffect 적합
#    - 최종 STM 모델에서 time effect만 시각화할 것이므로
#      통제변수도 함께 넣어 부분효과 형태로 추정
# ---------------------------------------------------------
if ("Origin_Country" %in% colnames(out3$meta) && "C_adj_i" %in% colnames(out3$meta)) {
  
  effect_time <- estimateEffect(
    1:K ~ s(Publication.Year) + Origin_Country + C_adj_i,
    stmobj = final_stm_20_adj,
    metadata = out3$meta,
    uncertainty = "Global"
  )
  
} else if ("Origin_Country" %in% colnames(out3$meta) && !"C_adj_i" %in% colnames(out3$meta)) {
  
  effect_time <- estimateEffect(
    1:K ~ s(Publication.Year) + Origin_Country,
    stmobj = final_stm_20_adj,
    metadata = out3$meta,
    uncertainty = "Global"
  )
  
} else if (!"Origin_Country" %in% colnames(out3$meta) && "C_adj_i" %in% colnames(out3$meta)) {
  
  effect_time <- estimateEffect(
    1:K ~ s(Publication.Year) + C_adj_i,
    stmobj = final_stm_20_adj,
    metadata = out3$meta,
    uncertainty = "Global"
  )
  
} else {
  
  effect_time <- estimateEffect(
    1:K ~ s(Publication.Year),
    stmobj = final_stm_20_adj,
    metadata = out3$meta,
    uncertainty = "Global"
  )
}

# 11. estimateEffect 기반 토픽별 시계열 선그래프 저장 함수
#     - stm::plot.estimateEffect를 직접 파일로 저장
#     - 95% CI 포함
# ---------------------------------------------------------
save_topic_ci_plot <- function(effect_obj, topic_num, topic_name, years, file_stub) {
  
  png(
    filename = paste0(file_stub, ".png"),
    width = 1400,
    height = 900,
    res = 180
  )
  
  plot(
    effect_obj,
    covariate = "Publication.Year",
    topics = topic_num,
    model = final_stm_20_adj,
    method = "continuous",
    xlab = "Publication Year",
    ylab = "Expected Topic Proportion",
    main = paste0(topic_name, " (T", topic_num, ")"),
    printlegend = FALSE,
    xaxt = "n",
    linecol = "navy",
    ci.level = 0.95
  )
  axis(1, at = years, labels = years)
  dev.off()
  
  pdf(
    file = paste0(file_stub, ".pdf"),
    width = 10,
    height = 6
  )
  plot(
    effect_obj,
    covariate = "Publication.Year",
    topics = topic_num,
    model = final_stm_20_adj,
    method = "continuous",
    xlab = "Publication Year",
    ylab = "Expected Topic Proportion",
    main = paste0(topic_name, " (T", topic_num, ")"),
    printlegend = FALSE,
    xaxt = "n",
    linecol = "navy",
    ci.level = 0.95
  )
  axis(1, at = years, labels = years)
  dev.off()
  
  print(paste("✅ 저장 완료:", paste0(file_stub, ".png / .pdf")))
}

save_topic_ci_plot <- function(effect_obj, topic_num, topic_name, years, file_stub, line_color = "navy") {
  
  png(
    filename = paste0(file_stub, ".png"),
    width = 1400,
    height = 900,
    res = 180
  )
  
  plot(
    effect_obj,
    covariate = "Publication.Year",
    topics = topic_num,
    model = final_stm_20_adj,
    method = "continuous",
    xlab = "Publication Year",
    ylab = "Expected Topic Proportion",
    main = paste0(topic_name, " (T", topic_num, ")"),
    printlegend = FALSE,
    xaxt = "n",
    linecol = line_color,
    ci.level = 0.95
  )
  axis(1, at = years, labels = years)
  dev.off()
  
  pdf(
    file = paste0(file_stub, ".pdf"),
    width = 10,
    height = 6
  )
  
  plot(
    effect_obj,
    covariate = "Publication.Year",
    topics = topic_num,
    model = final_stm_20_adj,
    method = "continuous",
    xlab = "Publication Year",
    ylab = "Expected Topic Proportion",
    main = paste0(topic_name, " (T", topic_num, ")"),
    printlegend = FALSE,
    xaxt = "n",
    linecol = line_color,
    ci.level = 0.95
  )
  axis(1, at = years, labels = years)
  dev.off()
  
  print(paste("✅ 저장 완료:", paste0(file_stub, ".png / .pdf")))
}

# K=20 기술 융합 생태계 매핑용 Edge Table 저장
# 기준: final_stm_20_adj / topicCorr / threshold = 0.15
# 라벨 없이, 토픽 번호와 5개 클러스터 정보만 저장
# =========================================================

# -------------------------------
# 0. 패키지 로드
# -------------------------------
library(stm)

# -------------------------------
# 1. 객체 확인
# -------------------------------
if (!exists("final_stm_20_adj")) stop("final_stm_20_adj 객체가 없습니다.")
if (!exists("mega_cluster_map")) stop("mega_cluster_map 객체가 없습니다.")

# -------------------------------
# 2. topicCorr 실행 및 상관행렬 추출
# -------------------------------
topic_corr_obj <- topicCorr(final_stm_20_adj)

if (!is.null(topic_corr_obj$posadj)) {
  corr_mat <- topic_corr_obj$posadj
} else if (!is.null(topic_corr_obj$cor)) {
  corr_mat <- topic_corr_obj$cor
} else {
  stop("topicCorr 결과에서 상관행렬을 찾을 수 없습니다.")
}

# -------------------------------
# 3. 토픽 이름 부여
# -------------------------------
K <- ncol(final_stm_20_adj$theta)

if (is.null(rownames(corr_mat))) rownames(corr_mat) <- paste0("T", 1:K)
if (is.null(colnames(corr_mat))) colnames(corr_mat) <- paste0("T", 1:K)

# -------------------------------
# 4. threshold 적용
# -------------------------------
threshold <- 0.15

adj_mat <- corr_mat
adj_mat[adj_mat < threshold] <- 0
diag(adj_mat) <- 0
adj_mat[adj_mat < 0] <- 0

# -------------------------------
# 5. upper triangle에서만 edge 추출
#    (중복 제거: T1-T2와 T2-T1 중 하나만 저장)
# -------------------------------
edge_idx <- which(adj_mat > 0 & upper.tri(adj_mat), arr.ind = TRUE)

edge_table <- data.frame(
  from = rownames(adj_mat)[edge_idx[, 1]],
  to   = colnames(adj_mat)[edge_idx[, 2]],
  weight = adj_mat[edge_idx],
  stringsAsFactors = FALSE
)

# -------------------------------
# 6. 보조 변수 추가
# -------------------------------
edge_table$abs_weight <- abs(edge_table$weight)
edge_table$sign <- ifelse(edge_table$weight > 0, "positive",
                          ifelse(edge_table$weight < 0, "negative", "zero"))

# -------------------------------
# 7. 5개 클러스터 정보 붙이기
#    mega_cluster_map은 T1~TK 순서라고 가정
# -------------------------------
cluster_df <- data.frame(
  topic = paste0("T", 1:K),
  cluster = mega_cluster_map,
  stringsAsFactors = FALSE
)

edge_table$from_cluster <- cluster_df$cluster[match(edge_table$from, cluster_df$topic)]
edge_table$to_cluster   <- cluster_df$cluster[match(edge_table$to, cluster_df$topic)]

edge_table$edge_type <- ifelse(edge_table$from_cluster == edge_table$to_cluster,
                               "within_cluster", "between_cluster")

# -------------------------------
# 8. 정렬
# -------------------------------
edge_table <- edge_table[order(-edge_table$weight), ]

# -------------------------------
# 9. 저장
# -------------------------------
write.csv(edge_table, "STM_Topic_Ecosystem_Edge_Table.csv", row.names = FALSE)

cat("✅ 저장 완료: STM_Topic_Ecosystem_Edge_Table.csv\n")
cat("총 edge 수:", nrow(edge_table), "\n")
print(head(edge_table, 10))
# 가장 강한 연결 상위 10개
head(edge_table, 10)

# 클러스터 내부 vs 클러스터 간 평균 연결강도
aggregate(weight ~ edge_type, data = edge_table, mean)

# 어떤 클러스터 쌍이 강하게 연결되는지
cluster_pair_summary <- aggregate(weight ~ from_cluster + to_cluster,
                                  data = edge_table,
                                  FUN = function(x) c(mean = mean(x), n = length(x)))

print(cluster_pair_summary)

# 4. k개 군집용 cut height 계산 함수
# ==========================================
get_cut_height <- function(hc, k){
  n <- length(hc$order)
  if(k < 2 || k > n){
    stop("k는 2 이상, 토픽 수 이하만 가능합니다.")
  }
  sort(hc$height, decreasing = TRUE)[k - 1]
}

# 3) 엣지(선) 굵기 및 진하기 설정 (NaN 에러 방지 처리)
if (ecount(g) > 0) {
  min_w <- min(E(g)$weight)
  max_w <- max(E(g)$weight)
  
  if (min_w == max_w) {
    # 모든 선의 가중치가 동일하거나 선이 1개뿐인 경우 기본값 적용
    E(g)$width <- 2
    E(g)$color <- rgb(0.4, 0.4, 0.4, alpha = 0.5)
  } else {
    # 가중치가 다양할 경우 0 ~ 1 사이로 정규화
    weight_norm <- (E(g)$weight - min_w) / (max_w - min_w)
    
    # 선 굵기 (0.5 ~ 4.5)
    E(g)$width <- 0.5 + (weight_norm * 4) 
    
    # 선 투명도 (0.15 ~ 0.85)
    alpha_values <- 0.15 + (weight_norm * 0.7)
    
    # 선 색상 적용
    E(g)$color <- rgb(0.4, 0.4, 0.4, alpha = alpha_values)
  }
}

  pivot_longer(
