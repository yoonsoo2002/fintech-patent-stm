# =========================================================
# 1. 8개 클러스터 체계 기반 메가 클러스터 재설정 (표 내용 반영)
# =========================================================
mega_cluster_map <- rep(NA, 20)

# A: 기업 재무 및 회계 관리
mega_cluster_map[c(4, 11)] <- "A"
# B: 맞춤형 보험 및 타겟 식별
mega_cluster_map[c(3, 18)] <- "B"
# C: AI 기반 신용/투자 분석
mega_cluster_map[c(9, 13, 17)] <- "C"
# D: 블록체인 기반 에너지 금융
mega_cluster_map[c(14, 19)] <- "D"
# E: 블록체인 및 디지털 자산
mega_cluster_map[c(1, 6, 8, 12)] <- "E"
# F: 보안 인증 및 규제 컴플라이언스
mega_cluster_map[c(7, 15)] <- "F"
# G: 뱅킹 및 대금 정산
mega_cluster_map[c(5, 10)] <- "G"
# H: O2O 상거래 및 간편 결제
mega_cluster_map[c(2, 16, 20)] <- "H"

# =========================================================
# 2. 네트워크 분석용 라이브러리 및 데이터 준비
# =========================================================
library(stm)
library(igraph)

setwd("C:/Users/gram/Documents/핀테크 특허 분석/k찾기")

# 상관관계 계산 (임계치 0.15)
topic_corr_obj <- topicCorr(final_stm_20_adj)
corr_mat <- if (!is.null(topic_corr_obj$posadj)) topic_corr_obj$posadj else topic_corr_obj$cor
K <- 20
colnames(corr_mat) <- rownames(corr_mat) <- paste0("T", 1:K)

threshold <- 0.15
adj_mat <- corr_mat
adj_mat[adj_mat < threshold] <- 0
diag(adj_mat) <- 0

g <- graph_from_adjacency_matrix(adj_mat, mode = "undirected", weighted = TRUE)
all_topic_names <- paste0("T", 1:K)
missing_nodes <- setdiff(all_topic_names, V(g)$name)
if (length(missing_nodes) > 0) g <- add_vertices(g, nv = length(missing_nodes), name = missing_nodes)
g <- permute(g, match(all_topic_names, V(g)$name))

# 노드 데이터프레임 구축
theta_mat <- final_stm_20_adj$theta
topic_prevalence <- colMeans(theta_mat)
node_df <- data.frame(
  topic = paste0("T", 1:K),
  prevalence = topic_prevalence,
  cluster_group = mega_cluster_map,
  stringsAsFactors = FALSE
)

# 중심성(Betweenness) 계산 및 추가
node_df$betweenness <- betweenness(g, normalized = TRUE)

# =========================================================
# 3. 색상 설정 (사진 속 지정된 HEX 코드 적용)
# =========================================================
cluster_levels <- c("A", "B", "C", "D", "E", "F", "G", "H")
cluster_colors <- c("#AEC6CF", "#FFD1BA", "#D8BFD8", "#C1E1C1", 
                    "#FDFD96", "#B0C4DE", "#AFEEEE", "#FFB6C1")
names(cluster_colors) <- cluster_levels

node_df$color <- cluster_colors[node_df$cluster_group]

# 범례에 표시할 구체적인 라벨
legend_labels <- c(
  "기업 재무 및 회계 관리", "맞춤형 보험 및 타겟 식별",
  "AI 기반 신용/투자 분석", "블록체인 기반 에너지 금융",
  "블록체인 및 디지털 자산", "보안 인증 및 규제 컴플라이언스",
  "뱅킹 및 대금 정산", "O2O 상거래 및 간편 결제"
)

# =========================================================
# 4. 시각화 속성 설정 (라벨 및 선 강도 조절, 에러 방지 반영)
# =========================================================

# 1) 노드 크기 계산
node_df$node_size <- 12 + (node_df$prevalence - min(node_df$prevalence)) * 23 / (max(node_df$prevalence) - min(node_df$prevalence))

# 2) 노드 속성 설정
V(g)$size <- node_df$node_size
V(g)$color <- node_df$color
V(g)$label <- paste0("T", 1:K) 
V(g)$label.cex <- 0.8
V(g)$label.color <- "black"
V(g)$frame.color <- "gray40"

# 3) 엣지(선) 굵기 및 진하기 설정 (NaN 에러 방지 처리)
if (ecount(g) > 0) {
  min_w <- min(E(g)$weight)
  max_w <- max(E(g)$weight)
  
  if (min_w == max_w) {
    # 모든 선의 가중치가 동일하거나 선이 1개뿐인 경우 기본값 적용
    E(g)$width <- 2
    E(g)$color <- rgb(0.4, 0.4, 0.4, alpha = 0.5)
  } else {
    # 가중치가 다양할 경우 0 ~ 1 사이로 정규화
    weight_norm <- (E(g)$weight - min_w) / (max_w - min_w)
    
    # 선 굵기 (0.5 ~ 4.5)
    E(g)$width <- 0.5 + (weight_norm * 4) 
    
    # 선 투명도 (0.15 ~ 0.85)
    alpha_values <- 0.15 + (weight_norm * 0.7)
    
    # 선 색상 적용
    E(g)$color <- rgb(0.4, 0.4, 0.4, alpha = alpha_values)
  }
}
# =========================================================
# 5. 그래프 파일 저장 (PNG & PDF)
# =========================================================
set.seed(2026)
lay <- layout_with_fr(g, niter = 1000)

# 1) PNG 저장
png(filename = "STM_Network_8_Clusters.png", width = 2800, height = 2000, res = 300)
par(mar = c(1, 1, 3, 1))
plot(g, layout = lay, main = "FinTech Technology Structure: 8-Cluster Map")
legend("topleft", legend = legend_labels, pt.bg = cluster_colors, 
       pch = 21, pt.cex = 1.8, bty = "n", cex = 0.9, title = "Clusters")
dev.off()

# 2) PDF 저장
pdf("STM_Network_8_Clusters.pdf", width = 11, height = 8)
par(mar = c(1, 1, 3, 1))
plot(g, layout = lay, main = "FinTech Technology Structure: 8-Cluster Map")
legend("topleft", legend = legend_labels, pt.bg = cluster_colors, 
       pch = 21, pt.cex = 1.5, bty = "n", cex = 0.8, title = "Clusters")
dev.off()

print("✅ 시각화 이미지 저장 완료: STM_Network_8_Clusters.png / pdf")

# =========================================================
# 6. 그래프 관련 수치 데이터 CSV로 전체 추출
# =========================================================

# 1) 노드(토픽) 속성 데이터 저장 (토픽 비중, 중심성, 소속 클러스터 등)
write.csv(node_df[order(as.numeric(gsub("T", "", node_df$topic))), ], 
          "Network_Nodes_Info.csv", row.names = FALSE)

# 2) 엣지(연결선) 데이터 저장 (어떤 토픽끼리 연결되었고 가중치가 얼마인지)
edge_df <- as_data_frame(g, what = "edges")
write.csv(edge_df, "Network_Edges_Info.csv", row.names = FALSE)

# 3) 중심성(Betweenness) 기준 정렬 데이터
centrality_sorted <- node_df[order(-node_df$betweenness), ]
write.csv(centrality_sorted, "Centrality_8_Clusters_FULL.csv", row.names = FALSE)
write.csv(head(centrality_sorted, 3), "Top3_Betweenness_Topics.csv", row.names = FALSE)

print("✅ 그래프 관련 수치 데이터 CSV 저장 완료")
